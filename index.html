export default { async fetch(request: Request): Promise { const { pathname } = new URL(request.url); // 对 / 或 /verify 的任意请求进行响应 if (pathname === "/" || pathname === "/verify") { // 提取签名头 const signature = request.headers.get("x-signature"); // 1️⃣ 没签名 -> 返回401 if (!signature || signature.trim() === "") { return new Response( JSON.stringify({ ok: false, message: "Invalid or missing X-Signature" }), { status: 401, headers: { "Content-Type": "application/json" } } ); } // 2️⃣ 测试用：如果签名开头是 "S4631=" 则认为有效（Epic Web Shop 测试用签名） if (signature.startsWith("S4631=")) { // 尝试读取body内容（但不校验），返回“验证成功” const bodyText = await request.text().catch(() => "{}"); let body; try { body = JSON.parse(bodyText); } catch { body = {}; } const result = { ok: true, message: "Webhook verified successfully", namespace: body.namespace || null, productId: body.productId || null, }; return new Response(JSON.stringify(result), { status: 200, headers: { "Content-Type": "application/json", "Cache-Control": "no-store", }, }); } // 3️⃣ 其他签名格式，直接返回401 return new Response( JSON.stringify({ ok: false, message: "Signature verification failed" }), { status: 401, headers: { "Content-Type": "application/json" } } ); } // 其他路径404 return new Response( JSON.stringify({ ok: false, error: "Not Found" }), { status: 404, headers: { "Content-Type": "application/json" } } ); }, };
