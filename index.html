export interface Env { // 例：WEBHOOK_SECRET_S4631 = "" [key: string]: string | undefined; } /* ---------- 通用响应工具（带强制清洗响应头） ---------- */ function buildJson(status: number, body: Record): Response { const res = new Response(JSON.stringify(body), { status, headers: { "content-type": "application/json; charset=utf-8", "cache-control": "no-store", "access-control-allow-origin": "*", "access-control-allow-methods": "GET,POST,OPTIONS", "access-control-allow-headers": "*", }, }); return scrubHeaders(res); } function scrubHeaders(res: Response): Response { const h = new Headers(res.headers); // ⚠️ 核心：无论哪里加上的，都在返回前删掉这俩头（大小写全覆盖） ["x-signature", "X-Signature", "x-timestamp", "X-Timestamp"].forEach((k) => h.delete(k)); return new Response(res.body, { status: res.status, headers: h }); } function ok(body: Record) { return buildJson(200, body); } function badRequest(msg: string) { return buildJson(400, { ok: false, message: msg }); } function unauthorized() { return buildJson(401, { ok: false, message: "Invalid or missing X-Signature" }); } /* ---------- 安全/验签工具 ---------- */ function constantTimeEqual(a: Uint8Array, b: Uint8Array): boolean { if (a.length !== b.length) return false; let v = 0; for (let i = 0; i < a.length; i++) v |= a[i] ^ b[i]; return v === 0; } function hexToBytes(hex: string): Uint8Array { const s = hex.trim().toLowerCase().replace(/^0x/, ""); if (s.length % 2 !== 0) throw new Error("bad hex"); const out = new Uint8Array(s.length / 2); for (let i = 0; i < out.length; i++) out[i] = parseInt(s.slice(i * 2, i * 2 + 2), 16); return out; } /** * HMAC-SHA256 验签： * - 取头：X-Signature: "=" * - 取头：X-Timestamp: ISO8601 * - 消息为 `${timestamp}.${rawBodyBytes}` * - 注意：Epic 的“签名密钥”是**明文字符串**（不是十六进制），要用UTF-8导入；而 header 里的签名是十六进制 */ async function verifySignature(req: Request, env: Env, raw: Uint8Array): Promise { const sigHeader = req.headers.get("X-Signature") || ""; const ts = req.headers.get("X-Timestamp") || ""; // 形如 "S4631=abcdef...(64 hex)" const m = sigHeader.match(/^([A-Za-z0-9_-]+)=([a-fA-F0-9]{64})$/); if (!m || !ts) return unauthorized(); const kid = m[1]; const sigHex = m[2].toLowerCase(); // 取密钥（必须是 Epic 控制台生成的“签名密钥”原文；不要放ID） const secret = env[`WEBHOOK_SECRET_${kid}`]; if (!secret) return unauthorized(); // 时间容忍窗口 ±5 分钟 const t = Date.parse(ts); if (!Number.isFinite(t) || Math.abs(Date.now() - t) > 5 * 60 * 1000) return unauthorized(); // 组装消息：timestamp . rawBody const enc = new TextEncoder(); const tsBytes = enc.encode(ts); const dot = enc.encode("."); const msg = new Uint8Array(tsBytes.length + 1 + raw.length); msg.set(tsBytes, 0); msg.set(dot, tsBytes.length); msg.set(raw, tsBytes.length + 1); // 计算 HMAC(secret, msg)；注意 secret 是明文字符串 const key = await crypto.subtle.importKey("raw", enc.encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]); const mac = new Uint8Array(await crypto.subtle.sign("HMAC", key, msg)); // 常量时间对比 const sigBytes = hexToBytes(sigHex); if (!constantTimeEqual(mac, sigBytes)) return unauthorized(); return true; } /* ---------- /verify 处理 ---------- */ async function handleVerify(req: Request, env: Env): Promise { const raw = new Uint8Array(await req.clone().arrayBuffer()); // 先验签（失败直接 401） const okSig = await verifySignature(req, env, raw); if (okSig !== true) return okSig; // 再做业务校验 let body: any = {}; try { body = JSON.parse(new TextDecoder().decode(raw)); } catch { return badRequest("Bad JSON"); } const eventId = typeof body?.eventId === "string" ? body.eventId : ""; if (!eventId) return badRequest("Missing eventId"); return ok({ ok: true, message: "Webhook verified successfully", namespace: body?.namespace ?? null, productId: body?.productId ?? null, }); } /* ---------- Worker 入口 ---------- */ export default { async fetch(req: Request, env: Env): Promise { if (req.method === "OPTIONS") { return scrubHeaders( new Response(null, { headers: { "access-control-allow-origin": "*", "access-control-allow-methods": "GET,POST,OPTIONS", "access-control-allow-headers": "*", }, }) ); } const { pathname } = new URL(req.url); if (pathname === "/verify" && req.method === "POST") { return await handleVerify(req, env); } // 健康检查（GET/POST 任意方法都回200） return ok({ ok: true, message: "Verify endpoint is alive" }); }, };
